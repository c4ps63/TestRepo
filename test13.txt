import { Component } from '@angular/core';

@Component({
  selector: 'app-counter',
  template: `
    <button (click)="dec()">-</button>
    <span>{{ count }}</span>
    <button (click)="inc()">+</button>
  `
})
export class CounterComponent {
  count = 0;

  inc() { this.count++; }
  dec() { this.count--; }
}
